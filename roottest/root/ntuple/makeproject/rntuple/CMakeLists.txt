# Generate common dictionary for custom class
ROOTTEST_GENERATE_DICTIONARY(
  rntuple_event_dict
  ${CMAKE_CURRENT_SOURCE_DIR}/ntuple_makeproject_header.h
  LINKDEF ${CMAKE_CURRENT_SOURCE_DIR}/ntuple_makeproject_header_LinkDef.h
  # Otherwise TFile::MakeProject will find the dictionary module and will not generate the corresponding class header
  NO_ROOTMAP NO_CXXMODULE
  FIXTURES_SETUP generated_event_dictionary_rntuple
)

# Write the RNTuple with the custom class field
ROOTTEST_GENERATE_EXECUTABLE(
  write_rntuple
  write_rntuple.cxx rntuple_event_dict.cxx
  LIBRARIES Core RIO ROOTNTuple
  FIXTURES_REQUIRED generated_event_dictionary_rntuple
  FIXTURES_SETUP write_rntuple_executable)

ROOTTEST_ADD_TEST(write_rntuple
                  EXEC ./write_rntuple
                  FIXTURES_REQUIRED write_rntuple_executable
                  FIXTURES_SETUP written_rntuple)

# Call MakeProject on the output file
ROOTTEST_GENERATE_EXECUTABLE(
  makeproject_rntuple
  makeproject_rntuple.cxx
  LIBRARIES Core RIO ROOTNTuple
  FIXTURES_SETUP makeproject_rntuple)

ROOTTEST_ADD_TEST(makeproject_rntuple
                  EXEC ./makeproject_rntuple
                  FIXTURES_REQUIRED "written_rntuple;makeproject_rntuple"
                  FIXTURES_SETUP makeproject_rntuple_called)

# Read back the class instance thanks to the shared library generated by MakeProject.
# Make sure it corresponds to the data stored in the original custom class
set(RNTUPLESTLTEST_DIR ${CMAKE_CURRENT_BINARY_DIR}/librntuplestltest)

# Like ROOTTEST_GENERATE_EXECUTABLE, link the sanitizer defaults in ASan builds
if(TARGET ROOTStaticSanitizerConfig)
  set(_sanitizer_config ${ROOT_LIBRARY_DIR}/${CMAKE_STATIC_LIBRARY_PREFIX}ROOTStaticSanitizerConfig${CMAKE_STATIC_LIBRARY_SUFFIX})
endif()

# Standalone project that builds read_rntuple. It is built by a test via
# "ctest --build-and-test", so it creates no target in this project and the
# test can run in parallel with the others.
set(_read_rntuple_project ${CMAKE_CURRENT_BINARY_DIR}/read_rntuple_project)
file(CONFIGURE OUTPUT ${_read_rntuple_project}/src/CMakeLists.txt @ONLY CONTENT [[
cmake_minimum_required(VERSION @CMAKE_MINIMUM_REQUIRED_VERSION@)
project(read_rntuple LANGUAGES CXX)
set(CMAKE_CXX_STANDARD @CMAKE_CXX_STANDARD@)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(read_rntuple "@CMAKE_CURRENT_SOURCE_DIR@/read_rntuple.cxx")

# For ROOT (don't use find_package(ROOT REQUIRED) here because it's slow)
target_include_directories(read_rntuple PRIVATE "@CMAKE_BINARY_DIR@/include")
target_link_directories(read_rntuple PRIVATE "@ROOT_LIBRARY_DIR@")

target_include_directories(read_rntuple PRIVATE "@CMAKE_CURRENT_BINARY_DIR@")
if(MSVC)
  # ROOT's headers need ROOT's compiler flags, e.g. -Zc:__cplusplus
  set(CMAKE_CXX_FLAGS "@CMAKE_CXX_FLAGS@")
  target_link_libraries(read_rntuple PRIVATE libCore libRIO libROOTNTuple librntuplestltest)
else()
  target_link_libraries(read_rntuple PRIVATE Core RIO ROOTNTuple rntuplestltest)
endif()
target_link_directories(read_rntuple PRIVATE "@RNTUPLESTLTEST_DIR@")

# Thread library and ASan flags, which ROOT's executables get from the main project
target_link_libraries(read_rntuple PRIVATE @CMAKE_THREAD_LIBS_INIT@ @_sanitizer_config@)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} @ASAN_EXTRA_CXX_FLAGS@")
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} @ASAN_EXTRA_EXE_LINKER_FLAGS@")

set_target_properties(read_rntuple PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "$<1:@CMAKE_CURRENT_BINARY_DIR@>"
  BUILD_RPATH "@RNTUPLESTLTEST_DIR@")
]])

# The read test source code links against the library generated by MakeProject,
# called in the previous test. Therefore, we always have to recompile at test
# time. The most convenient ROOT macro to do this is the ROOT_ADD_TEST with the
# BUILD option, which builds a CMake project and runs a command.
if(MSVC)
  set(_env ENVIRONMENT PATH=${RNTUPLESTLTEST_DIR})
endif()
ROOTTEST_TARGETNAME_FROM_FILE(read_test_name read_rntuple)
ROOT_ADD_TEST(${read_test_name}
  COMMAND ./read_rntuple
  WORKING_DIR ${CMAKE_CURRENT_BINARY_DIR}
  BUILD read_rntuple
  SOURCE_DIR ${_read_rntuple_project}/src
  BINARY_DIR ${_read_rntuple_project}/build
  FIXTURES_REQUIRED makeproject_rntuple_called
  ${_env}
)
