cmake_minimum_required(VERSION 3.20)
project(CheckCUDA LANGUAGES C CXX)

list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}" "${ROOT_SOURCE_DIR}/cmake/modules")
# Keep repeated CTest runs independent of options cached by the previous configure.
foreach(opt cuda tmva-gpu all minimal gminimal CMAKE_CUDA_COMPILER)
  unset(${opt} CACHE)
endforeach()
set(fortran OFF)
set(expect_cuda OFF)
set(cuda_available ON)
set(expect_cuda_option OFF)
set(expect_tmva_gpu_option OFF)

if(CASE STREQUAL "disabled" OR CASE STREQUAL "cached-disabled" OR CASE STREQUAL "all-disabled")
  set(cuda OFF CACHE BOOL "")
  set(tmva-gpu OFF CACHE BOOL "")
  if(CASE STREQUAL "cached-disabled")
    set(CMAKE_CUDA_COMPILER mock-cuda-compiler CACHE FILEPATH "")
  elseif(CASE STREQUAL "all-disabled")
    set(all ON)
  endif()
elseif(CASE STREQUAL "cuda")
  set(cuda ON CACHE BOOL "")
  set(expect_cuda ON)
  set(expect_cuda_option ON)
elseif(CASE STREQUAL "tmva-gpu")
  set(tmva-gpu ON CACHE BOOL "")
  set(expect_cuda ON)
  set(expect_tmva_gpu_option ON)
elseif(CASE STREQUAL "all" OR CASE STREQUAL "all-cuda-disabled" OR CASE STREQUAL "all-tmva-disabled")
  set(all ON)
  set(expect_cuda ON)
  set(expect_cuda_option ON)
  set(expect_tmva_gpu_option ON)
  if(CASE STREQUAL "all-cuda-disabled")
    set(cuda OFF CACHE BOOL "")
    set(expect_cuda_option OFF)
  elseif(CASE STREQUAL "all-tmva-disabled")
    set(tmva-gpu OFF CACHE BOOL "")
    set(expect_tmva_gpu_option OFF)
  endif()
elseif(CASE STREQUAL "minimal")
  set(minimal ON)
elseif(CASE STREQUAL "gminimal" OR CASE STREQUAL "all-gminimal")
  set(gminimal ON)
  if(CASE STREQUAL "all-gminimal")
    set(all ON)
  endif()
elseif(CASE STREQUAL "unavailable")
  set(all ON)
  set(expect_cuda ON)
  set(cuda_available OFF)
elseif(NOT CASE STREQUAL "default")
  message(FATAL_ERROR "Unknown test case: ${CASE}")
endif()

include(CheckCompiler)
include(RootBuildOptions)

if(expect_cuda AND NOT cuda_checked)
  message(FATAL_ERROR "CUDA compiler detection was skipped")
endif()
if(expect_cuda AND cuda_available AND NOT cuda_enabled)
  message(FATAL_ERROR "The CUDA language was not initialized")
endif()
if(NOT "${cuda}" STREQUAL "${expect_cuda_option}")
  message(FATAL_ERROR "Expected cuda=${expect_cuda_option}, got ${cuda}")
endif()
if(NOT "${tmva-gpu}" STREQUAL "${expect_tmva_gpu_option}")
  message(FATAL_ERROR "Expected tmva-gpu=${expect_tmva_gpu_option}, got ${tmva-gpu}")
endif()
